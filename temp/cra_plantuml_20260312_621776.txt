' ============= MAIN AGREEMENT TABLE =============
entity corporate_reporting_agreement {
    *agreement_id: BIGINT <<PK>> <<generated>>
    --
    agreement_name: NVARCHAR(200)
    customer_id: NVARCHAR(50)
    status_code: NVARCHAR(20) <<FK>>
    pricing_order_ref: NVARCHAR(100) <<unique>>
    pricing_type: NVARCHAR(20)
    fixed_price: DECIMAL(15,2)
    version_number: INT
    parent_agreement_id: BIGINT <<FK>>
    is_current_version: BIT
    submitted_date: DATETIME
    approved_date: DATETIME
    effective_date: DATE
    expiry_date: DATE
    cart_expiry_date: DATETIME
    created_by: NVARCHAR(100)
    created_date: DATETIME
    last_updated_by: NVARCHAR(100)
    last_updated_date: DATETIME
}

' ============= JUNCTION TABLES =============
entity agreement_product_part {
    *agreement_product_part_id: BIGINT <<PK>> <<generated>>
    --
    agreement_id: BIGINT <<FK>>
    product_part_code: NVARCHAR(30) <<FK>>
    created_date: DATETIME
}

entity contact {
    *contact_id: BIGINT <<PK>> <<generated>>
    --
    customer_id: NVARCHAR(50)
    contact_name: NVARCHAR(200)
    email: NVARCHAR(200)
    is_active: BIT
    created_date: DATETIME
    last_updated_date: DATETIME

    *agreement_contact_id: BIGINT <<PK>> <<generated>>
    --
    agreement_id: BIGINT <<FK>>

    created_date: DATETIME
}

entity recipient_product_part {
    *recipient_product_part_id: BIGINT <<PK>> <<generated>>

    recipient_id: BIGINT <<FK>>

    created_date: DATETIME
